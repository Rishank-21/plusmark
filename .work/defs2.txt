  if not pg_catalog.pg_try_advisory_xact_lock(pg_catalog.hashtext('plusmark.refresh_rollups')) then
    return pg_catalog.jsonb_build_object('skipped', true);
  end if;

  select coalesce(pg_catalog.array_agg(x.local_day order by x.local_day desc), '{}') into v_dirty

  foreach v_day in array v_dirty loop
    perform internal.rollup_build_days(v_day, v_day);

  delete from ext.rollup_dirty_day where local_day >= v_from or local_day = any (v_dirty);
  return pg_catalog.jsonb_build_object('skipped', false, 'from_date', v_from, 'timezone', v_tz,
    'dirty_days', pg_catalog.cardinality(v_dirty));
